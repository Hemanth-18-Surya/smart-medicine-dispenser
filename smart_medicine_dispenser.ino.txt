// --- Function Declarations ---
void updateDisplay(struct tm timeinfo);
void checkReminders(struct tm timeinfo);
void triggerReminder(int index);

void setup() {
  Serial.begin(115200);

  pinMode(BUZZER, OUTPUT);
  digitalWrite(BUZZER, 1);

  // Initialize Sensors & Servo Motors
  for (int i = 0; i < 3; i++) {
    pinMode(ledPins[i], OUTPUT);
    digitalWrite(ledPins[i], LOW);
    pinMode(irTakenPins[i], INPUT);

    servos[i].attach(servoPins[i]);
    servos[i].write(0); // Default closed position
  }

  // LCD Initialization (4-bit mode)
  lcd.begin(16, 2);
  lcd.clear();
  lcd.setCursor(0, 0);
  lcd.print("SMART MEDICINE");
  lcd.setCursor(0, 1);
  lcd.print("SYSTEM STARTING");
  delay(2000);

  // Wi-Fi Connection
  WiFi.begin(WIFI_SSID, WIFI_PASSWORD);
  lcd.clear();
  lcd.print("Connecting WiFi...");
  
  int attempts = 0;
  while (WiFi.status() != WL_CONNECTED && attempts < 20) {
    delay(500);
    attempts++;
  }

  lcd.clear();
  if (WiFi.status() == WL_CONNECTED) {
    lcd.setCursor(0, 0);
    lcd.print("WiFi Connected!");
    lcd.setCursor(0, 1);
    lcd.print(WiFi.localIP());
    delay(5000);

    // Sync Internet Time via NTP
    configTime(gmtOffset_sec, daylightOffset_sec, ntpServer);
    
    lcd.clear();
    lcd.print("Syncing Time...");
    struct tm timeinfo;
    time_t now = time(nullptr);
    while (now < 24 * 3600 && attempts < 40) {
      delay(500);
      now = time(nullptr);
      attempts++;
    }
  } else {
    lcd.setCursor(0, 0);
    lcd.print("WiFi Failed!");
    lcd.setCursor(0, 1);
    lcd.print("Check Credentials");
    delay(3000);
  }

  // Web Server Endpoints
  server.on("/", HTTP_GET, []() {
    server.send(200, "text/html", index_html);
  });

  server.on("/status", HTTP_GET, []() {
    struct tm timeinfo;
    char timeBuff[9] = "--:--:--";
    if (getLocalTime(&timeinfo)) {
      snprintf(timeBuff, sizeof(timeBuff), "%02d:%02d:%02d", timeinfo.tm_hour, timeinfo.tm_min, timeinfo.tm_sec);
    }

    String json = "{\"time\":\"" + String(timeBuff) + 
                  "\",\"status\":[" + String(medStatus[0]) + "," + String(medStatus[1]) + "," + String(medStatus[2]) + "]}";
    server.send(200, "application/json", json);
  });

  server.on("/setReminder", HTTP_GET, []() {
    if (server.hasArg("id") && server.hasArg("time")) {
      int id = server.arg("id").toInt() - 1;
      String t = server.arg("time");
      if (id >= 0 && id < 3 && t.length() == 5) {
        remHours[id] = t.substring(0, 2).toInt();
        remMinutes[id] = t.substring(3, 5).toInt();
        medStatus[id] = 0; // Reset status to pending when updated
        server.send(200, "text/plain", "OK");
        return;
      }
    }
    server.send(400, "text/plain", "Bad Request");
  });

  server.begin();
}

void loop() {
  server.handleClient();

  struct tm timeinfo;
  if (getLocalTime(&timeinfo)) {
    updateDisplay(timeinfo);
    checkReminders(timeinfo);

    // Reset daily status at midnight (00:00:00)
    if (timeinfo.tm_hour == 0 && timeinfo.tm_min == 0 && timeinfo.tm_sec == 0) {
      for (int i = 0; i < 3; i++) {
        alerted[i] = false;
        medStatus[i] = 0;
      }
    }

    // Re-arm alert triggers when reminder minute passes
    if (timeinfo.tm_sec == 0) {
      for (int i = 0; i < 3; i++) {
        if (remHours[i] != -1 && timeinfo.tm_min != remMinutes[i]) {
          alerted[i] = false;
        }
      }
    }
  }

  delay(200);
}

String getShortTime(int index) {
  if (remHours[index] == -1 || remMinutes[index] == -1) return "--:--";
  char buf[6];
  snprintf(buf, sizeof(buf), "%02d:%02d", remHours[index], remMinutes[index]);
  return String(buf);
}

void updateDisplay(struct tm timeinfo) {
  lcd.setCursor(0, 0);
  char timeStr[17];
  snprintf(timeStr, sizeof(timeStr), "TIME: %02d:%02d:%02d ", 
           timeinfo.tm_hour, timeinfo.tm_min, timeinfo.tm_sec);
  lcd.print(timeStr);

  lcd.setCursor(0, 1);
  char remStr[17];
  snprintf(remStr, sizeof(remStr), "R:%s %s %s", 
           getShortTime(0).c_str(), 
           getShortTime(1).c_str(), 
           getShortTime(2).c_str());
  lcd.print(remStr);
}

void checkReminders(struct tm timeinfo) {
  for (int i = 0; i < 3; i++) {
    if (remHours[i] != -1 && remMinutes[i] != -1) {
      if (timeinfo.tm_hour == remHours[i] && timeinfo.tm_min == remMinutes[i] && !alerted[i]) {
        triggerReminder(i);
        alerted[i] = true;
      }
    }
  }
}

void triggerReminder(int index) {
  bool medicineTaken = false;

  // Display alert on LCD
  lcd.clear();
  lcd.setCursor(0, 0);
  lcd.print("TAKE MEDICINE ");
  lcd.print(index + 1);

  // Turn ON LED indicator
  digitalWrite(ledPins[index], HIGH);

  // 1. Servo Action: Open (0 -> 90 degrees)
  for (int pos = 0; pos <= 90; pos++) {
    servos[index].write(pos);
    delay(15);
  }

  // 2. Hold box open for 15 Seconds (Alert active with Buzzer & IR checking)
  unsigned long servoWait = millis();
  while (millis() - servoWait < 15000) {
    server.handleClient();

    // Toggle Buzzer Beep
    digitalWrite(BUZZER, LOW);
    delay(150);
    digitalWrite(BUZZER, HIGH);
    delay(150);

    // Check IR Sensor for hand/medicine pickup (Active LOW)
    if (digitalRead(irTakenPins[index]) == LOW) {
      medicineTaken = true;
      break;
    }
  }

  // 3. Servo Action: Close (90 -> 0 degrees)
  for (int pos = 90; pos >= 0; pos--) {
    servos[index].write(pos);
    delay(15);
  }

  // Turn OFF LED and Buzzer
  digitalWrite(ledPins[index], LOW);
  digitalWrite(BUZZER, 1);

  // 4. Update status on LCD
  if (medicineTaken) {
    medStatus[index] = 1; // TAKEN
    lcd.setCursor(0, 1);
    lcd.print("STATUS: TAKEN  ");
  } else {m
    medStatus[index] = 2; // MISSED
    lcd.setCursor(0, 1);
    lcd.print("STATUS: MISSED ");
  }
  delay(2000);
  lcd.clear();
}
